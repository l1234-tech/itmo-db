## 4. Связь «Перелистывание» между героями и книгами (M:N)

    CREATE TABLE перелистывание (
        id_героя INTEGER REFERENCES герой(id_героя),
        id_книги INTEGER REFERENCES книга(id_книги),
        PRIMARY KEY (id_героя, id_книги)
    );

    INSERT INTO перелистывание (id_героя, id_книги) VALUES (1, 1);
    INSERT INTO перелистывание (id_героя, id_книги) VALUES (1, 2);
    INSERT INTO перелистывание (id_героя, id_книги) VALUES (2, 1);

## 5. Связь «Собирание» между героями и островками

    CREATE TABLE собирание (
        id_героя INTEGER REFERENCES герой(id_героя),
        id_островок INTEGER REFERENCES островок(id_островка),
        вид_добычи VARCHAR(30) NOT NULL
            CHECK (вид_добычи IN ('моллюски', 'устрицы')),
        PRIMARY KEY (id_героя, id_островок, вид_добычи)
    );

    INSERT INTO собирание (id_героя, id_островок, вид_добычи)
    VALUES (1, 1, 'моллюски');
    INSERT INTO собирание (id_героя, id_островок, вид_добычи)
    VALUES (2, 1, 'устрицы');
