## 3. EXISTS: студенты вечерней формы без отчества

    SELECT EXISTS (
        SELECT 1
        FROM Н_ЛЮДИ
        JOIN Н_ОБУЧЕНИЯ ON Н_ЛЮДИ.ИД = Н_ОБУЧЕНИЯ.ЧЛВК_ИД
        JOIN Н_УЧЕНИКИ ON Н_ОБУЧЕНИЯ.ЧЛВК_ИД = Н_УЧЕНИКИ.ЧЛВК_ИД
        JOIN Н_ПЛАНЫ ON Н_УЧЕНИКИ.ПЛАН_ИД = Н_ПЛАНЫ.ИД
        JOIN Н_ФОРМЫ_ОБУЧЕНИЯ ON Н_ПЛАНЫ.ФО_ИД = Н_ФОРМЫ_ОБУЧЕНИЯ.ИД
        WHERE Н_ФОРМЫ_ОБУЧЕНИЯ.НАИМЕНОВАНИЕ = 'Очно-заочная(вечерняя)'
          AND (Н_ЛЮДИ.ОТЧЕСТВО IS NULL OR Н_ЛЮДИ.ОТЧЕСТВО = '')
    );

Проверка из отчёта:

    SELECT COUNT(*) FROM Н_ЛЮДИ
    WHERE ОТЧЕСТВО IS NULL OR ОТЧЕСТВО = '';

В отчёте отмечено, что людей без отчества в Н_ЛЮДИ не найдено; EXISTS возвращает false.

## 4. Планы, по которым более двух заочных групп

    SELECT COUNT(Н_ГРУППЫ_ПЛАНОВ.ГРУППА) AS количество_групп,
           Н_ФОРМЫ_ОБУЧЕНИЯ.НАИМЕНОВАНИЕ,
           Н_ГРУППЫ_ПЛАНОВ.ПЛАН_ИД
    FROM Н_ГРУППЫ_ПЛАНОВ
    JOIN Н_ПЛАНЫ ON Н_ГРУППЫ_ПЛАНОВ.ПЛАН_ИД = Н_ПЛАНЫ.ИД

## 5. Средний возраст студентов по группам

    SELECT Н_ГРУППЫ_ПЛАНОВ.ГРУППА,
           AVG(EXTRACT(YEAR FROM AGE(Н_ЛЮДИ.ДАТА_РОЖДЕНИЯ))) AS средний_возраст
    FROM Н_ГРУППЫ_ПЛАНОВ
    JOIN Н_УЧЕНИКИ ON Н_ГРУППЫ_ПЛАНОВ.ПЛАН_ИД = Н_УЧЕНИКИ.ПЛАН_ИД
    JOIN Н_ОБУЧЕНИЯ ON Н_УЧЕНИКИ.ЧЛВК_ИД = Н_ОБУЧЕНИЯ.ЧЛВК_ИД
    JOIN Н_ЛЮДИ ON Н_ОБУЧЕНИЯ.ЧЛВК_ИД = Н_ЛЮДИ.ИД
    GROUP BY Н_ГРУППЫ_ПЛАНОВ.ГРУППА
    HAVING AVG(EXTRACT(YEAR FROM AGE(Н_ЛЮДИ.ДАТА_РОЖДЕНИЯ))) > (
        SELECT MIN(EXTRACT(YEAR FROM AGE(Н_ЛЮДИ.ДАТА_РОЖДЕНИЯ)))
        FROM Н_ЛЮДИ
        JOIN Н_ОБУЧЕНИЯ ON Н_ЛЮДИ.ИД = Н_ОБУЧЕНИЯ.ЧЛВК_ИД
        JOIN Н_УЧЕНИКИ ON Н_ОБУЧЕНИЯ.ЧЛВК_ИД = Н_УЧЕНИКИ.ЧЛВК_ИД
        JOIN Н_ГРУППЫ_ПЛАНОВ ON Н_УЧЕНИКИ.ПЛАН_ИД = Н_ГРУППЫ_ПЛАНОВ.ПЛАН_ИД
        WHERE Н_ГРУППЫ_ПЛАНОВ.ГРУППА = '1100'
    );

## 6. Отчисленные до 01.09.2012 студенты «Программной инженерии»

    SELECT Н_УЧЕНИКИ.ГРУППА, Н_ЛЮДИ.ИД, Н_ЛЮДИ.ФАМИЛИЯ,
           Н_ЛЮДИ.ИМЯ, Н_ЛЮДИ.ОТЧЕСТВО,
           Н_УЧЕНИКИ.П_ПРКОК_ИД AS номер_пункта_приказа
    FROM Н_НАПР_СПЕЦ
    JOIN Н_НАПРАВЛЕНИЯ_СПЕЦИАЛ
        ON Н_НАПР_СПЕЦ.ИД = Н_НАПРАВЛЕНИЯ_СПЕЦИАЛ.НС_ИД
    JOIN Н_ПЛАНЫ ON Н_НАПРАВЛЕНИЯ_СПЕЦИАЛ.ИД = Н_ПЛАНЫ.НАПС_ИД
    JOIN Н_ФОРМЫ_ОБУЧЕНИЯ ON Н_ПЛАНЫ.ФО_ИД = Н_ФОРМЫ_ОБУЧЕНИЯ.ИД
    JOIN Н_УЧЕНИКИ ON Н_ПЛАНЫ.ИД = Н_УЧЕНИКИ.ПЛАН_ИД
    JOIN Н_ЛЮДИ ON Н_УЧЕНИКИ.ЧЛВК_ИД = Н_ЛЮДИ.ИД
    WHERE Н_УЧЕНИКИ.КОНЕЦ_ПО_ПРИКАЗУ < DATE '2012-09-01'
      AND Н_УЧЕНИКИ.КОНЕЦ = Н_УЧЕНИКИ.КОНЕЦ_ПО_ПРИКАЗУ
      AND Н_ФОРМЫ_ОБУЧЕНИЯ.НАИМЕНОВАНИЕ IN ('Очная', 'Заочная')
      AND Н_НАПР_СПЕЦ.НАИМЕНОВАНИЕ = 'Программная инженерия'
    ORDER BY Н_УЧЕНИКИ.ГРУППА, Н_ЛЮДИ.ФАМИЛИЯ, Н_ЛЮДИ.ИМЯ;

## Дополнительное задание к защите: рекурсивная иерархия подразделений

Для представления иерархии подразделений используем таблицу Н_ОТДЕЛЫ. Предполагается, что ИД идентифицирует подразделение, а ОТД_ИД хранит ИД его родительского подразделения. Сначала нужно проверить, что эта связь подтверждена ограничением базы, и выяснить, каким образом обозначаются корневые подразделения.

### Полный запрос

    WITH RECURSIVE иерархия AS (
        -- 1. Начальная часть: корневые подразделения
        SELECT
            o.ИД,
            o.ОТД_ИД,
            o.КОРОТКОЕ_ИМЯ,
            0 AS ГЛУБИНА,
            COALESCE(
                NULLIF(o.ИМЯ_В_ИМИН_ПАДЕЖЕ, ''),
                o.КОРОТКОЕ_ИМЯ
            )::TEXT AS ПОЛНЫЙ_ПУТЬ,
            ARRAY[o.ИД] AS ПОСЕЩЁННЫЕ
        FROM Н_ОТДЕЛЫ AS o
        WHERE o.ОТД_ИД IS NULL

        UNION ALL

        -- 2. Рекурсивная часть: находим детей найденных подразделений
        SELECT
            child.ИД,
            child.ОТД_ИД,
            child.КОРОТКОЕ_ИМЯ,
            parent.ГЛУБИНА + 1,
            parent.ПОЛНЫЙ_ПУТЬ || ' -> ' ||
                COALESCE(
                    NULLIF(child.ИМЯ_В_ИМИН_ПАДЕЖЕ, ''),
                    child.КОРОТКОЕ_ИМЯ
                ),
            parent.ПОСЕЩЁННЫЕ || child.ИД
        FROM Н_ОТДЕЛЫ AS child
        JOIN иерархия AS parent
            ON child.ОТД_ИД = parent.ИД
        WHERE NOT (child.ИД = ANY(parent.ПОСЕЩЁННЫЕ))
    )
    SELECT
        ИД,
        ОТД_ИД,
        КОРОТКОЕ_ИМЯ,
        ПОЛНЫЙ_ПУТЬ,
        ГЛУБИНА
    FROM иерархия
    ORDER BY ПОЛНЫЙ_ПУТЬ;

### Что делает каждая часть

1. WITH RECURSIVE иерархия AS (...) объявляет рекурсивный CTE — промежуточный результат, который будет расширяться на каждом шаге.
2. Начальная часть выбирает строки, у которых ОТД_ИД IS NULL. Это предполагаемые корни. Для них глубина равна нулю, а путь начинается с названия самого подразделения.
3. UNION ALL объединяет корневые записи с записями, найденными рекурсивной частью.
4. JOIN иерархия AS parent ON child.ОТД_ИД = parent.ИД ищет дочерние записи: у ребёнка в ОТД_ИД записан идентификатор уже найденного родителя.
5. parent.ГЛУБИНА + 1 увеличивает глубину на каждом уровне. При принятом соглашении корень имеет глубину 0, его ребёнок — 1, внук — 2.
6. parent.ПОЛНЫЙ_ПУТЬ || ' -> ' || ... дописывает название ребёнка к пути родителя. Оператор || соединяет текстовые строки.
7. COALESCE(NULLIF(ИМЯ_В_ИМИН_ПАДЕЖЕ, ''), КОРОТКОЕ_ИМЯ) выбирает имя в именительном падеже, если оно не NULL и не пустое; иначе используется короткое имя.
8. ARRAY[o.ИД] создаёт массив посещённых ID. Выражение parent.ПОСЕЩЁННЫЕ || child.ИД добавляет к массиву текущий ID, а NOT (child.ИД = ANY(parent.ПОСЕЩЁННЫЕ)) не даёт повторно посетить один идентификатор на этом пути при наличии цикла.
9. Внешний SELECT выводит итоговые строки, а ORDER BY ПОЛНЫЙ_ПУТЬ сортирует их по собранным путям.

### Проверки перед запуском

Проверь, как обозначаются корни:

    SELECT ИД, ОТД_ИД, КОРОТКОЕ_ИМЯ
    FROM Н_ОТДЕЛЫ
    WHERE ОТД_ИД IS NULL;

Проверь внешние ключи таблицы (имя таблицы должно быть доступно через текущий search_path):

    SELECT conname, pg_get_constraintdef(oid)
    FROM pg_constraint
    WHERE conrelid = 'н_отделы'::regclass
      AND contype = 'f';

Если нет корней с ОТД_ИД IS NULL, результат рекурсивного запроса может оказаться пустым. Необходимо проверить реальные данные и ограничения, прежде чем менять стартовое условие.

### Как адаптировать для другой таблицы

Если иерархическая связь хранится в другой таблице, меняются:
- исходная таблица после FROM;
- первичный идентификатор элемента;
- столбец родительского ID;
- название элемента для пути;
- стартовое условие, которое определяет корни.

Главная конструкция остаётся такой: начальная выборка корней, UNION ALL, затем JOIN исходной таблицы с CTE по связи «родительский ID ребёнка = ID найденного родителя».
