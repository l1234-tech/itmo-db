## 1. LEFT JOIN оценок и ведомостей

В исходном отчёте фильтры рассматриваются попарно, так как ЧЛВК_ИД = 105590 и ЧЛВК_ИД > 153285 не могут быть истинны одновременно.

    -- Условия a и b
    SELECT Н_ОЦЕНКИ.ПРИМЕЧАНИЕ, Н_ВЕДОМОСТИ.ЧЛВК_ИД
    FROM Н_ОЦЕНКИ
    LEFT JOIN Н_ВЕДОМОСТИ
        ON Н_ОЦЕНКИ.КОД = Н_ВЕДОМОСТИ.ОЦЕНКА
    WHERE Н_ОЦЕНКИ.ПРИМЕЧАНИЕ > 'неявка'
      AND Н_ВЕДОМОСТИ.ЧЛВК_ИД = 105590;

    -- Условия a и c
    SELECT Н_ОЦЕНКИ.ПРИМЕЧАНИЕ, Н_ВЕДОМОСТИ.ЧЛВК_ИД

        ON Н_ОЦЕНКИ.КОД = Н_ВЕДОМОСТИ.ОЦЕНКА

## 2. RIGHT JOIN людей, обучений и учеников

    SELECT Н_ЛЮДИ.ИД, Н_ОБУЧЕНИЯ.ЧЛВК_ИД, Н_УЧЕНИКИ.ИД
    FROM Н_ЛЮДИ
    RIGHT JOIN Н_ОБУЧЕНИЯ
        ON Н_ЛЮДИ.ИД = Н_ОБУЧЕНИЯ.ЧЛВК_ИД
    RIGHT JOIN Н_УЧЕНИКИ
        ON Н_ОБУЧЕНИЯ.ЧЛВК_ИД = Н_УЧЕНИКИ.ЧЛВК_ИД
    WHERE Н_ЛЮДИ.ФАМИЛИЯ > 'Соколов'
      AND Н_ОБУЧЕНИЯ.НЗК = '999080';

В исходном отчёте запрос вернул 0 строк. Указано, что для записи с НЗК 999080 фамилия — «Жуков», поэтому фильтр фамилии не проходит в порядке сравнения, принятом в отчёте.

## 3. EXISTS: студенты вечерней формы без отчества

    SELECT EXISTS (
        SELECT 1
        FROM Н_ЛЮДИ
        JOIN Н_ОБУЧЕНИЯ ON Н_ЛЮДИ.ИД = Н_ОБУЧЕНИЯ.ЧЛВК_ИД
        JOIN Н_УЧЕНИКИ ON Н_ОБУЧЕНИЯ.ЧЛВК_ИД = Н_УЧЕНИКИ.ЧЛВК_ИД
        JOIN Н_ПЛАНЫ ON Н_УЧЕНИКИ.ПЛАН_ИД = Н_ПЛАНЫ.ИД
        JOIN Н_ФОРМЫ_ОБУЧЕНИЯ ON Н_ПЛАНЫ.ФО_ИД = Н_ФОРМЫ_ОБУЧЕНИЯ.ИД
        WHERE Н_ФОРМЫ_ОБУЧЕНИЯ.НАИМЕНОВАНИЕ = 'Очно-заочная(вечерняя)'
          AND (Н_ЛЮДИ.ОТЧЕСТВО IS NULL OR Н_ЛЮДИ.ОТЧЕСТВО = '')
    );

Проверка из отчёта:

    SELECT COUNT(*) FROM Н_ЛЮДИ
    WHERE ОТЧЕСТВО IS NULL OR ОТЧЕСТВО = '';

В отчёте отмечено, что людей без отчества в Н_ЛЮДИ не найдено; EXISTS возвращает false.

## 4. Планы, по которым более двух заочных групп

    SELECT COUNT(Н_ГРУППЫ_ПЛАНОВ.ГРУППА) AS количество_групп,
           Н_ФОРМЫ_ОБУЧЕНИЯ.НАИМЕНОВАНИЕ,
           Н_ГРУППЫ_ПЛАНОВ.ПЛАН_ИД
    FROM Н_ГРУППЫ_ПЛАНОВ
    JOIN Н_ПЛАНЫ ON Н_ГРУППЫ_ПЛАНОВ.ПЛАН_ИД = Н_ПЛАНЫ.ИД
    JOIN Н_ФОРМЫ_ОБУЧЕНИЯ ON Н_ПЛАНЫ.ФО_ИД = Н_ФОРМЫ_ОБУЧЕНИЯ.ИД
    WHERE Н_ФОРМЫ_ОБУЧЕНИЯ.НАИМЕНОВАНИЕ = 'Заочная'
    GROUP BY Н_ГРУППЫ_ПЛАНОВ.ПЛАН_ИД, Н_ФОРМЫ_ОБУЧЕНИЯ.НАИМЕНОВАНИЕ
    HAVING COUNT(Н_ГРУППЫ_ПЛАНОВ.ГРУППА) > 2;

Проверочный запрос из отчёта показывает, что групп с заочной формой в используемых данных нет, поэтому результат пустой.

## 5. Средний возраст студентов по группам

    SELECT Н_ГРУППЫ_ПЛАНОВ.ГРУППА,
           AVG(EXTRACT(YEAR FROM AGE(Н_ЛЮДИ.ДАТА_РОЖДЕНИЯ))) AS средний_возраст
    FROM Н_ГРУППЫ_ПЛАНОВ
    JOIN Н_УЧЕНИКИ ON Н_ГРУППЫ_ПЛАНОВ.ПЛАН_ИД = Н_УЧЕНИКИ.ПЛАН_ИД
    JOIN Н_ОБУЧЕНИЯ ON Н_УЧЕНИКИ.ЧЛВК_ИД = Н_ОБУЧЕНИЯ.ЧЛВК_ИД
    JOIN Н_ЛЮДИ ON Н_ОБУЧЕНИЯ.ЧЛВК_ИД = Н_ЛЮДИ.ИД
    GROUP BY Н_ГРУППЫ_ПЛАНОВ.ГРУППА
    HAVING AVG(EXTRACT(YEAR FROM AGE(Н_ЛЮДИ.ДАТА_РОЖДЕНИЯ))) > (
        SELECT MIN(EXTRACT(YEAR FROM AGE(Н_ЛЮДИ.ДАТА_РОЖДЕНИЯ)))
        FROM Н_ЛЮДИ
        JOIN Н_ОБУЧЕНИЯ ON Н_ЛЮДИ.ИД = Н_ОБУЧЕНИЯ.ЧЛВК_ИД
        JOIN Н_УЧЕНИКИ ON Н_ОБУЧЕНИЯ.ЧЛВК_ИД = Н_УЧЕНИКИ.ЧЛВК_ИД
        JOIN Н_ГРУППЫ_ПЛАНОВ ON Н_УЧЕНИКИ.ПЛАН_ИД = Н_ГРУППЫ_ПЛАНОВ.ПЛАН_ИД
        WHERE Н_ГРУППЫ_ПЛАНОВ.ГРУППА = '1100'
    );

## 6. Отчисленные до 01.09.2012 студенты «Программной инженерии»

    SELECT Н_УЧЕНИКИ.ГРУППА, Н_ЛЮДИ.ИД, Н_ЛЮДИ.ФАМИЛИЯ,
           Н_ЛЮДИ.ИМЯ, Н_ЛЮДИ.ОТЧЕСТВО,
           Н_УЧЕНИКИ.П_ПРКОК_ИД AS номер_пункта_приказа
    FROM Н_НАПР_СПЕЦ
    JOIN Н_НАПРАВЛЕНИЯ_СПЕЦИАЛ
        ON Н_НАПР_СПЕЦ.ИД = Н_НАПРАВЛЕНИЯ_СПЕЦИАЛ.НС_ИД
    JOIN Н_ПЛАНЫ ON Н_НАПРАВЛЕНИЯ_СПЕЦИАЛ.ИД = Н_ПЛАНЫ.НАПС_ИД
    JOIN Н_ФОРМЫ_ОБУЧЕНИЯ ON Н_ПЛАНЫ.ФО_ИД = Н_ФОРМЫ_ОБУЧЕНИЯ.ИД
    JOIN Н_УЧЕНИКИ ON Н_ПЛАНЫ.ИД = Н_УЧЕНИКИ.ПЛАН_ИД
    JOIN Н_ЛЮДИ ON Н_УЧЕНИКИ.ЧЛВК_ИД = Н_ЛЮДИ.ИД
    WHERE Н_УЧЕНИКИ.КОНЕЦ_ПО_ПРИКАЗУ < DATE '2012-09-01'
      AND Н_УЧЕНИКИ.КОНЕЦ = Н_УЧЕНИКИ.КОНЕЦ_ПО_ПРИКАЗУ
      AND Н_ФОРМЫ_ОБУЧЕНИЯ.НАИМЕНОВАНИЕ IN ('Очная', 'Заочная')
      AND Н_НАПР_СПЕЦ.НАИМЕНОВАНИЕ = 'Программная инженерия'
    ORDER BY Н_УЧЕНИКИ.ГРУППА, Н_ЛЮДИ.ФАМИЛИЯ, Н_ЛЮДИ.ИМЯ;

## 7. Люди, не являющиеся студентами ФКТИУ (без DISTINCT)

    SELECT Н_ЛЮДИ.ИД, Н_ЛЮДИ.ФАМИЛИЯ, Н_ЛЮДИ.ИМЯ, Н_ЛЮДИ.ОТЧЕСТВО
    FROM Н_ЛЮДИ
    WHERE NOT EXISTS (
        SELECT 1
        FROM Н_УЧЕНИКИ
        JOIN Н_ПЛАНЫ ON Н_УЧЕНИКИ.ПЛАН_ИД = Н_ПЛАНЫ.ИД
        JOIN Н_ОТДЕЛЫ ON Н_ПЛАНЫ.ОТД_ИД = Н_ОТДЕЛЫ.ИД
        WHERE Н_УЧЕНИКИ.ЧЛВК_ИД = Н_ЛЮДИ.ИД
          AND Н_ОТДЕЛЫ.КОРОТКОЕ_ИМЯ = 'КТиУ'
    );

## Дополнительное задание к защите: рекурсивная иерархия подразделений

В схеме БД «Учебный процесс» для примера иерархии используется таблица `Н_ОТДЕЛЫ`: `ИД` — идентификатор подразделения, `ОТД_ИД` — предполагаемый идентификатор родительского подразделения. Решение предполагает, что корневые записи имеют `ОТД_ИД IS NULL`; это следует проверить в конкретном экземпляре базы.

    WITH RECURSIVE иерархия AS (
        SELECT
            o.ИД,
            o.ОТД_ИД,
            o.КОРОТКОЕ_ИМЯ,
            o.ИМЯ_В_ИМИН_ПАДЕЖЕ,
            0 AS ГЛУБИНА,
            COALESCE(
                NULLIF(o.ИМЯ_В_ИМИН_ПАДЕЖЕ, ''),
                o.КОРОТКОЕ_ИМЯ
            )::TEXT AS ПОЛНЫЙ_ПУТЬ,
            ARRAY[o.ИД] AS ПОСЕЩЁННЫЕ
        FROM Н_ОТДЕЛЫ AS o
        WHERE o.ОТД_ИД IS NULL

        UNION ALL

        SELECT
            child.ИД,
            child.ОТД_ИД,
            child.КОРОТКОЕ_ИМЯ,
            child.ИМЯ_В_ИМИН_ПАДЕЖЕ,
            parent.ГЛУБИНА + 1,
            parent.ПОЛНЫЙ_ПУТЬ || ' -> ' ||
                COALESCE(
                    NULLIF(child.ИМЯ_В_ИМИН_ПАДЕЖЕ, ''),
                    child.КОРОТКОЕ_ИМЯ
                ),
            parent.ПОСЕЩЁННЫЕ || child.ИД
        FROM Н_ОТДЕЛЫ AS child
        JOIN иерархия AS parent
            ON child.ОТД_ИД = parent.ИД
        WHERE NOT (child.ИД = ANY(parent.ПОСЕЩЁННЫЕ))
    )
    SELECT ИД, ОТД_ИД, КОРОТКОЕ_ИМЯ, ПОЛНЫЙ_ПУТЬ, ГЛУБИНА
    FROM иерархия
    ORDER BY ПОЛНЫЙ_ПУТЬ;

Запрос начинает обход с корневых подразделений, затем на каждом шаге находит дочерние записи по условию `child.ОТД_ИД = parent.ИД`. Полный путь дополняется именем дочернего подразделения, глубина увеличивается на единицу. Массив посещённых ID защищает обход от повторного посещения узлов при наличии циклов в данных.
